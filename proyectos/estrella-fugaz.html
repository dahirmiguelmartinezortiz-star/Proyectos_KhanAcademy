// Primera estrella (arriba a la izquierda)
let xPos = 0;
let yPos = 0;

// Segunda estrella (arriba a la derecha)
let xPos2 = 400;
let yPos2 = 50;

function setup() {
  let canvas = createCanvas(400, 400);
  canvas.parent('sketch-container');
  noStroke();
}

function draw() {
  background(29, 40, 115);

  // Primera estrella
  fill(255, 242, 0);
  ellipse(xPos, yPos, 10, 10);

  // Segunda estrella
  fill(255, 255, 255);
  ellipse(xPos2, yPos2, 8, 8);

  // Mover primera estrella hacia abajo a la derecha
  xPos = xPos + 3;
  yPos = yPos + 2;

  // Mover segunda estrella hacia abajo a la izquierda
  xPos2 = xPos2 - 2;
  yPos2 = yPos2 + 3;
}

function resetAnimation() {
  xPos = 0;
  yPos = 0;
  xPos2 = 400;
  yPos2 = 50;
  loop();
}
