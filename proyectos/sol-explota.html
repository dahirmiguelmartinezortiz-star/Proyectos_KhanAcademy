let sunSize = 30;

function setup() {
  let canvas = createCanvas(400, 400);
  canvas.parent('sketch-container');
  noStroke();
}

function draw() {
  // El cielo azul
  background(82, 222, 240);

  // El sol en el horizonte
  fill(255, 204, 0);
  ellipse(200, 298, sunSize, sunSize);

  // La tierra tapando la mitad del sol
  fill(76, 168, 67);
  rect(0, 300, 400, 100);

  sunSize = sunSize + 1;
}

function resetAnimation() {
  sunSize = 30;
  loop();
}
